cmake_minimum_required(VERSION 3.24)
project(Squeeze VERSION 1.0.1 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(Squeeze WIN32 src/app.cpp src/engine.cpp assets/app.rc)
target_compile_definitions(Squeeze PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(Squeeze PRIVATE src generated)
target_link_libraries(Squeeze PRIVATE d2d1 dwrite windowscodecs ole32 shell32 comdlg32 dwmapi dxgi bcrypt cabinet comctl32)
if(MSVC)
  target_compile_options(Squeeze PRIVATE /W4 /permissive- /utf-8 /O1 /GL /Gy)
  target_link_options(Squeeze PRIVATE /LTCG /OPT:REF /OPT:ICF /MANIFEST:NO)
endif()
add_executable(SqueezeTests tests/core_tests.cpp src/engine.cpp assets/app.rc)
target_compile_definitions(SqueezeTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(SqueezeTests PRIVATE src generated)
target_link_libraries(SqueezeTests PRIVATE ole32 shell32 dxgi bcrypt cabinet)
if(MSVC)
  target_compile_options(SqueezeTests PRIVATE /utf-8 /W4)
endif()
enable_testing()
add_test(NAME core COMMAND SqueezeTests)
